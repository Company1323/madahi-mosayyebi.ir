«یَا ثامِن ُالائمّه»
➖➖➖➖➖

ای تاجدار خاور یا ثامِنُ الائمّه
فرمانروای اغبر یا ثامِن ُالائمّه
مصباح چرخ اخضر یا ثامِن ُالائمّه
مرآت وجه دلبر یا ثامِن ُالائمّه
کشتیّ دینه لنگر یا ثامِن ُالائمّه
ایرانیانه مفخر یا ثامِن ُالائمّه
تندیس وَقر حیدر یا ثامِن ُالائمّه
*****
تنویر نور حقدن بدرُالدُّجیٰ یاراندون               
هفت آسمانعشقه شمسُ الضُّحیٰ یاراندون               
مَرْضیّ امر داور عینُ الرّضا یاراندون               
وجهوندی رشک جنّت بس دلربا یاراندون               
اولدون طبیب عالم درد آشنا یاراندون               
مشکل گشاعلی تک مشکل گشا یاراندون               
               درده آدون مسخّر یا ثامن الائمه
*****
لوح وقارون اوخشار هر آیه ؛ سیر نوره               
ترویج اولار کمالون آفاقه سوره سوره               
فخر ایلیر حریمون معناده دشت طوره               
صحنون سالار کلیمی سینا چولونده شوره               
بیر بزم طاعت اولموش انسان وجنّ و حوره               
طوف ایلین مزارون والله چاتار سُروره               
               سسلر آدون مکرّر یا ثامن الائمه
*****
ساینده گردیش ایلور چرخ زمانه مولا               
سایه سالوب رواقون فرق جهانه مولا               
فخر ایلیَر مزارون هفت آسمانه مولا               
حِلّ و حریم قبرون اولموش یگانه مولا               
گردون دولاندی کام ایرانیانه مولا               
فاخردی بارگاهون خلد و جنانه مولا               
               قبرون فداسی گوزلر یا ثامن الائمه
*****
سندندی یابن زهرا دنیانین آبروسی               
هفت آسمان و چرخ خضرانین آبروسی               
صبح بیاض و لیل یلدانین آبروسی               
نوح و خلیل و هود وعیسیٰ نین آبروسی               
آرتوب نفس لروندن قرآنین آبروسی               
حور ِ قصور و باغ رضوانین آبروسی               
               ای وامدار کوثر یا ثامن الائمه
*****
ای آبروی هستی وی افتخار شیعه               
فخر ایلیوب سنیلن پروردگار شیعه               
مست کرامتوندور چشم خمار شیعه               
سندن چاتوبدی اوجه شأن و وقار شیعه               
آدون کی واردی سینماز بو اقتدار شیعه               
ایکّی جهاندا سن سن دار و ندار شیعه               
               ای شیعیانه سرور یا ثامن الائمه
*****
عشقوندی رمز ایمان سیر ِ کمالْ حدّی               
دریای اقتدارین سندندی جزر و مدّی               
احقاقیوه ئورکدن هر کیم عقیده گتدی               
اول سیل اعتقادی سیندیردی وار نه سدّی               
اعمال ، هر نه اولسا مقبول یا کی ردّی               
سندن قالان ولایت اقبالینه سند دی               
               ای هشت و چاره زیور یا ثامن الائمه
*****
طرز بیان آلوبسان پیغمبرین ئوزوندن               
شیوا سۆزون آلوب خط خیرالورا سۆزوندن               
آلمیش گوزون طراوت شاه عرب گوزوندن               
تاپمیش یوزون وجاهت ختم الرُسل یوزوندن               
عالم دم آلدی سیزدن شامات وگوندوزوندن               
آزماز یولون گئدر کیم بیله یولون دوزوندن               
               ایلر بو ذکری ازبر یا ثامن الائمه
*****
تنویر رحمتونده بس آتشین شرر وار               
شاهانه مرقدونده بیر عالم دگر وار               
کرّوبیان بیلوللر اول قبریده نلر وار               
صدق دیلیله باخسان دلداریدن خبر وار               
عشقونده یا رضاجان عالمجه دربه در وار               
وصل حریم یاره بس قانلی سینه لر وار               
               عشقیله سسلنلّر یا ثامن الائمه
*****
بیر سلسله ملکدور دورینده خاکسارون               
حوری وَشان جنت قبرونده بی قرارون               
هرباصفا کی گوز وار صحنونده اشکبارون               
جاندان گئچن گوزللر همّیشه جان نثارون               
عشقون سالوبدی حَصره هرسینه نین حصارون               
طاغوتیانی خوفه سالمیش بو اقتدارون               
               کفره آدون مسخّر یا ثامن الائمه
*****
گون باشیوه گرکدورتا وارجهان دولانسون               
قدّوسیان عالم ساینده دالدالانسون               
نار محبتیندن نکتار باده یانسون               
رهرو ولایتونده تا واردی جان دایانسون               
شهد امامتوندن عالملره جالانسون               
مأمونیان عالم منکوب اولوب تالانسون               
               شهباز عشقه شهپر یا ثامن الائمه
*****
گلشن گرک برات بارانی سندن آلسون               
انسان گرک کماله فرمانی سندن جآلسون               
عاشق گرک اصول شایانی سندن آلسون               
قطره تجلّیات دریانی سندن آلسون               
اسرار دینه شیعه برهانی سندن آلسون               
ایتگین شمیم قبر زهرانی سندن آلسون               
               زهرایه سبط اطهر یا ثامن الائمه
*****
تا کشتی ولانین طوفاندا لنگری وار               
تا بادۀ وفانین عالمده ساغری وار               
تا بارگاه عشقون گویلرده داوری وار               
تا برج هل اتیٰ نین اون ایکّی اختری وار               
تا کنز لافتیٰ نین تابنده گوهری وار               
والله رضا آدیندا هر لحظه مشتری وار               
               فرمانروای کشور یا ثامن الائمه
*****
هر باشدا یابن زهرا بیر فکر وبیرهوا وار               
محرابیده مجزّا اذکار وار صلا وار               
گلمیش نوایه نی تک هر دیلده بیر نوا وار               
اما منیم باشیمدا بیر سرّ برملا وار               
مدّ نظرده آقا بیر گنبد طلا وار               
بیر گنبدی که عرش حقدن اونا جلا وار               
               ائت کامیمی معطر یا ثامن الائمه
*****
وجهون ، شرف گویونده تابنده کوکبیدی               
آفاقیده رضایه آدون ملقّبیدی               
خورشید ابتهاجون زینت ده ِ شبی دی               
اذکار رأفتونله جانلار لبالبیدی               
پرچم کی مرقدونده وار رکن مذهبیدی               
بو بس کی یادگار غوغای زینبیدی               
               جانلار فداسی یکسر یا ثامن الائمه
*****
ای آدی مین تسلیٰ هر درد بی دوایه               
وئرّوخ سنی قسم بیز اولاد مصطفایه               
سندن بو نوعی مولا شیعه ائدور گلایه               
اوُل واسطه بیزیلن درگاه کبریایه               
گوندر بو جمع زاری بیر وقفه کربلایه               
بین الحرمده نی تک گلسون هامی نوایه               
               دلدن دییاخ مکرّر یا ثامن الائمه
*****
قورتارسؤزی«مُصَیـّب»قوی سۆزقالادیلینده               
شأن رضایه لایق چون تحفه یوخ الینده               
باشون اشاقه سالما اِکرام محفلینده               
بحر رضاده غرق اول چوخ قالما ساحلینده               
رفع ایلیَر گر اولسا هر درد و نیسگیلینده               
بوشرطیلن کی اولسون صدق و صفا دلینده               
               گوزیاشلی سسله سـَن گر یا ثامن الائمه
