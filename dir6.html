خروشان هَزارِ مُحن زینبم
ولا باده سیندن ایچن زینبم
حسین عشقینه روح وتن زینبم
یولوندا جانیندان گئچن زینبم
یزیدی فنایه چکن زینبم
اونون تختینی تتره دن زینبم
سپهره سالان لرزه ؛ من زینبم
*****
من اجلاس عشقین تولّاسیَم
گلستان دینون صفا یاسیم
بو خلقین همان اشجع الناسیم
علی نین درخشنده الماسیَم
حسینون بیلون ؛ آیری عبّاسیم
                ابوالفضل تک صف شکن زینبم
                سپهره سالان لرزه ؛ من زینبم
*****
منم فارِسِ پهنۀ نینوا
منم دختر یل یل لافتی
منم فاتح قلب خیرالنسا
یزیدین ائدن سرّینی برملا
اونی مسندیندن سالان توپراقا
                اونون باشینه کول سَپَن زینبم
                سپهره سالان لرزه ؛ من زینبم
*****
بولور خلق ؛ معنای قرآن منم
بولور قطره سیل خروشان منم
یزیده دئیون شیر میدان منم
او بیر مور دور چون سلیمان منم
یغیشسین بساطی کی سلطان منم
                تانور من کیمم کیمسه دن زینبم
                سپهره سالان لرزه ؛ من زینبم
*****
منم روح انّا فَتحنا بیلون
منم شأن فتحاً مُبینا بیلون
منم سرّ یاسین و طاها بیلون
منم روح و جسم تولا بیلون
منم نصّ قرآنه معنا بیلون
                حدیث ولانی بُولن زینبم
                سپهره سالان لرزه ؛ من زینبم
*****
منم بعدِ عباس پرچم ووران
منم شور و احساسیدن دم ووران
منم چاره سیز درده مرهم ووران
بساط سلاطینی برهم ووران
یزید آغزینا مشت محکم ووران
                تانورسیز منی شیرزن زینبم
                سپهره سالان لرزه ؛ من زینبم
*****
بیلون؛عشق زینب وداد اوسته دی
اونون مسلکی اعتقاد اوسته دی
یزیدین سوزی لیک عناداوسته دی
کی شخصیتی پول وآد اوسته دی
اونون کاخی ظلم وفساداوسته دی
                بو کاخی بنادن سوکن زینبم
                سپهره سالان لرزه ؛ من زینبم
*****
من عنوان عشقه چاتان زینبم
کمک سیز آدام سیز قالان زینبم
گئدوب کوفه شهرین آلان زینبم
همان کوفیه اود سالان زینبم
همان، محمله باش چالان زینبم
                یازاللار غریب وطن زینبم
                سپهره سالان لرزه ؛ من زینبم
*****
منیم خطبه م آفاقی مبهوت ائدر
سوزوم حقدی،حقْ قبض طاغوت ائدر
یزیدین تبارین علی الموت ائدر
سالار تختیدن لال ومسکوت ائدر
یزیده همان تختی تابوت ائدر
                نفاقه باشین اگمین زینبم
                سپهره سالان لرزه ؛ من زینبم
*****
علی نین قیزین ائتدیلر دربه در
خدا ؛ قلب زینب هاچان دینجلر
اگرچه جفادن اولوب خونجگر
ولی پرچمی تا قیامت اسر
زنا اهلی نین اصل و نسلین کسر
                یولون مرتضانین گئدن زینبم
                سپهره سالان لرزه ؛ من زینبم
*****
منم کربلانین بلاسین گورن
منم قتلگه ماجراسین گورن
او مقتلده شمرین جفاسین گورن
او یئرده حسینون صفاسین گورن
بوغازیندا خنجر یاراسین گورن
                حسینون بوغازین ئوپن زینبم
                سپهره سالان لرزه ؛ من زینبم
*****
بیر عده یازاللار من آواریم
بیر عده دیئللر کی بیچاریم
دیئر بعضیلر؛ چوخ گونی قاریم
بولن یوخ ولی دردلره چاریم
فقط بیر باش عشقینده آواریم
                «مُصیّب»سوزون ثبت ائدن زینبم
                سپهره سالان لرزه ؛ من زینبم
