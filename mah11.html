ای شاه مظلومان لبیک حسین لبیک
جانلار سنه قربان لبیک حسین لبیک
*****
سویله عزاداره دردی اولانلاره              
             صدق و صفا دلده گلسون بو درباره
دردین دئسون شاه و قولسوز علمداره             
             ایلَلـلَه واللّهی هر دردینه چاره
*****
ای شیعیه حجّت وی قائم نهضت             
             سن بیل آنان جانی وئر عشقیمه قیمت
بو بی ریا جَمعه ایله ئوزون قسمت             
             گوندر مزاریندان بیر نامۀ دعوت
*****
شمس وقمر سسلرلبیک حسین لبیک             
             ریح و مَطَر سسلر لبیک حسین لبیک
بو سینه زن دسته عشق ومرام اوسته             
             شام و سحر سسلر لبیک حسین لبیک
*****
ای قانینه غلطان قانه باتان عطشان              
کان سخاوت سن عالملره سلطان             
بیز سائل کویون سن معدن احسان             
             قووما بیزی شاها درگاه پاکیندان
*****
ای فخر ائدن یرده طفل لب عطشانه             
             قانلی بوغازیندا اوچ شعبه پیکانه
سندن بو عاشقلر ایستور یانا یانه             
             علقم کناریندا قورسون عزاخانه
*****
مست ائت بیزی مولا جام وثاقیندن             
             قانه باتوب گوزلر قبرین فراقیندن
احسان ائله آقا لطف و سیاقیندن             
             گلسون بو جمعیت ئوپسون رواقیندن
*****
سبط یَدُاللّهه بیزدن سلام اولسون             
             باشسیز قالان شاهه بیزدن سلام اولسون
سِرّ فَدَیْناهه بیزدن سلام اولسون             
             اول قانلی درگاهه بیزدن سلام اولسون
*****
ای سبط پیغمبر وی معرفت کانی             
             سندن آلار شیعه هر درده درمانی
وئرروخ قسم مولا سن بیل آنان جانی             
             کسمه بوهیأتدن سن لطفیوی آنی
*****
جرم و خطا بیزدن مولا عطا سندن             
             سؤز و بُکا بیزدن قلبه صفا سندن
ذکر و دعا بیزدن جود و سخا سندن             
             مین آه و وا بیزدن بیر کربلا سندن
*****
دلده صفا اولدوق درگاهیوه سائل             
             کبر و ریانی بیز ائتدوق آقا زائل
گل اول بو اخلاصه ارزش ئوزون قائل             
             ایله بیزی مولا اعزازیوه نائل
*****
لطف و عطا ایستر گویده ملک سندن             
             دریاده احسانین آلمیش سَمک سندن
ال بو «مصیّب» دن شاها اتک سندن             
             هاردا دوشر درده ایستر کمک سندن
