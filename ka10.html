«شهادت امام حسن(ع)»
➖➖➖➖➖

مرغ خوشخوان رازینی پنهان صبادن ساخلاماز                
               آشنا سرّین خفاده آشنادن ساخلاماز
بلبل ، عالی نغمۀ مستانه سین گلزاریده               
               اختفاده یاس و نسرین و همادن ساخلاماز
بستۀ زنجیر زلف یار ، سوز آهینی               
               سینه ده گیزلین نگار مه لقادن ساخلاماز
خال اگر چه صورتین حُسن و جمالین آرتیرار               
               لیک رخساری خدنگ ابتلادن ساخلاماز
رسم عالم دور وفا اهلی سوزون یاره دیئر               
               سینه ده سرّین وفالی ، بی وفادن ساخلاماز
یار ایللر یاریلن همدوش و همخواب اولسادا               
               چرخ گردون قلبینی تیر خطادن ساخلاماز
سنگ خارا آفتابیلن اگر هم توش اولا               
               قلب خاراسی همان نور و ضیادن ساخلاماز
سودۀ زهریله الماس وجودی دوغرانان               
               شیشۀ تقدیرینی سنگ قضادن ساخلاماز
بیر سبوی آب سیراب اولسا زهر نابیلن               
               دیل دوداقی یاندیرار قلبی نوادن ساخلاماز
رسم دوران دور کی صرصر گلسه گولزاری پوزار               
               نقش فتّان هئچ گولی باد فنادن ساخلاماز
هانسی گولدور حُسنِ طلعتده حَسَن تک بی بدیل               
               بیر گوزل کی خلقی لفظ مرحبادن ساخلاماز
همجوار اولدی بو گول ایللر مغیلان خاریلن               
               خارِ بنیانکن ئوزون نیش جفادن ساخلاماز

گور نئجه بیر قطره پوزدی رحمتین دریاسینی
آرتیروب آل علی نین محنت عظماسینی

جُعده مسموم ایلیوبدور اشبه پیغمبری               
               سالدی شمسِ هل اتایه سایۀ اغواسینی
نیله میشدی یارینه آخر سلیل مرتضی               
               قاسمین قویدی آتاسیز شاد ائدوب اعداسینی
مکریلن آدون قویان دنیاده خال المومنین               
               تولیوبدور جعده نی اول هنده نین همتاسینی
وئردی بیر کیسه قیزیل ، آلدی اونون ایمانینی               
               کوزۀ آبه توکوبدور سودۀ الماسینی
نیمۀ شب مجتبی دُوردی مُشوَّش ، خوابیدن               
               سو ایچوب رفع ایله سین مولامیز استسقاسینی
کوزه نی یتجاق دوداقه گور نئدوب آب سبو               
                رخسارین پوزوب الدن آلوب یاراسینی
ایچدی سو اما ئورک اودلاندی آرتیردی عطش               
               صرصرجور اسدی صولدورسون ولایت یاسینی
نور عین فاطمه ئولماقینه آگاه اولوب                
               ال آتوب اوغدی ئورک باشین،یانان اعضاسینی
اللرین قویدی یئره دورسین ایاقه مجتبی               
               گوردی دیز گلمور فلک بوکموش قد رعناسینی
آگه اولدی کوزه ایچره آب یوخ زهرآب وار               
               بولدی کیم سیراب ائدوبدور زهریلن صهباسینی
سسلنوبدور زینبین امداده حال زاریلن               
               دوردی زینب خوابیدن آلجاق اونون نجواسینی
گوردی تا قارداش دوداقینده باجی قان لخته سین               
               ووردی باشه قوزیوبدور عرشه آه و واسینی
اضطراب بی نهایت زینبین قلبین آلوب               
               راه چاره آختاروب غمدن قوتارسین یاسینی
سسلنوب قارداش سنی کیمسه بواحواله سالوب               
               کیم ایاقدان سالدی آیا زینبین طاهاسینی
سویلیوبدور غمکشیده زینبه مظلوم حسن               
               علتین آلما خبر کیم بیلمه سین آیاسینی
بیلمسون جَرْیانی عبدالله و بَکر و قاسمیم               
               نه حسین بیلسون نه آگاه ائت اونون عبّاسینی
آگه اولسا ماجرادن گر ابوالفضلیم باجی               
               تیغ جانفرساسی دوغرار قاتلین اعضاسینی
لیک تنها بیر اشارت ائتدی آب کوزیه               
               زینب آلدی مطلبین سرّین ،سوزون ایماسینی
بیرده سسلندی ؛ باجی ایله حسینی باخبر               
               تا گله بالینیمه بیرده گوروم سیماسینی
سویله قارداشین اولوب زهریله قارداش خونجگر               
               مجتبانین درک ائله ساعات وانفساسینی
ترک ائدوب تا منزلی زینب ،چاغوردی مجتبی               
               جعده نی ، اول قاتل دون مایۀ سفلی سینی
تترین لبلر دگوبدور بیربیره ؛ ای بی وفا               
               نیله میشدیم من سنه ییخدین ولا عنقاسینی
داغدار ائتدون علی نی روضۀ رضوانیده               
               ایلیوبسن باغریقان پیغمبرین زهراسینی
الغرض گلدی حسین قارداشینون بالینینه               
               بیلمورم گوردی نه نوعیله گل حمراسینی
اختیارین وئردی الدن ووردی باشیندن آقام               
               سیل اولان گوزیاشی دوتدی پهنۀ سیماسینی
تا بیلوب آب سبو اول دیل دوداقه نیلیوب               
               کوزه نی آلدی اله فاش ایله سین احساسینی
ایسته دی یتسون دوداقه کوزۀ آبی حسین               
               مجتبی ایلوب مخاطب زینب کبراسینی
سسلنوبدور آل سبونی سن حسینیمدن باجی               
               قویماایچسون اول سودان یاندیرماسین سوداسینی
من سوایچد یم سینه مه وُردی بو سو داغ فراق               
               قارداشیم امّا ائدر عطشان ، وصال امضاسینی
مجتبانین قسمتیندیدی بو صهبای بلا               
               میل ائدر امّا حسینیم کربلا صهباسینی
اوندا کی خنجر دمیندن نوش ائدر قان تشنه لب               
               چشمه چشمه قان دوتار درد و بلا صحراسینی
ماسوانی شوق وصل یاریلن باشدان آتار               
               بیر چوخور یئرده ائدر آللهیلن نجواسینی
گون قباقینده قوم اوسته پیکری عریان یاتار               
               قیزلاری عرشه سالار فریاد واویلاسینی
اوندا کی باشسیز بدن آتلار ایاقینده قالار               
               چاک چاک ایلَلّه سبط احمدین اعضاسینی
اوندا کی گون کربلاده نیزه باشیندان چیخار               
               اوندا درک ائت نهضتین سن وقفۀ حسّاسینی
هاله تک ،دوش، نی ده کی خورشیدعشقین آردیجا               
               قویما تا یاددان چیخا غوغای عاشوراسینی
هرنه گورسن سن بو یولدا ناامید اولما باجی               
               بجدلین ظلمین و یا شمر و سَنان بلواسینی
ایندی من سرمست جام وصل حیّ ذوالمنم               
               سینم اودلاندی ولی آلد یم شرف قَرطاسینی*
عازمم کوی وفاده واصل جانان اولام               
               حق ئوزی یاخشی تانور ئوز وصلینین شیداسینی
من کی جان وئردیم؛ باجی اولادیمه اول سرپناه               
               قویما پوزسون صرصر غم مجتبا اجلاسینی
بیر«مصیّب» سائلیم وار گرچه غرق سیّـئات               
               گوسترر امّا روان شعرینده ئوز احساسینی


* لوح ، کاغذ
